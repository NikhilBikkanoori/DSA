import java.util.*;
class Fancy {
    long mod=(long)1e9+7;
    ArrayList<Long> arr;
    long mul=1;
    long add=0;
    public Fancy() {
        arr=new ArrayList<>();
    }
    public void append(int val) {
        long v=(val-add)%mod;
        if(v<0) v+=mod;
        v=(v*modInverse(mul))%mod;
        arr.add(v);
    }
    public void addAll(int inc) {
        add=(add+inc)%mod;
    }
    public void multAll(int m) {
        mul=(mul*m)%mod;
        add=(add*m)%mod;
    }
    public int getIndex(int idx) {
        if(idx>=arr.size()) return -1;
        long val=arr.get(idx);
        return (int)((val*mul %mod+add) %mod);
    }
    private long modInverse(long x) {
        return power(x,mod-2);
    }
    private long power(long a,long b) {
        long res=1;
        a%=mod;
        while(b>0){
            if((b&1)==1)
                res=(res*a)%mod;
            a=(a*a)%mod;
            b>>= 1;
        }
        return res;
    }
}
