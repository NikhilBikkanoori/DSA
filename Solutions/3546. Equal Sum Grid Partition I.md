class Solution {
    public boolean canPartitionGrid(int[][] grid) {
        int n=grid.length;
        int m=grid[0].length;
        long arr[]=new long[n];
        long arr2[]=new long[n];
        long res[] = new long[m];
        long res2[] = new long[m];
        for(int i=0;i<n;i++){
            long val=0;
            for(int j=0;j<m;j++){
                val+=grid[i][j];
            }
            arr[i]=val;
        }
        long val=0;
        for(int i=n-1;i>=0;i--){
            val+=arr[i];
            arr2[i]=val;
        }
        for(int i=0;i<m;i++){
            val=0;
            for(int j=0;j<n;j++){
                val+=grid[j][i];
            }
            res[i]=val;
        }
        val=0;
        for(int i=m-1;i>=0;i--){
            val+=res[i];
            res2[i]=val;
        }
        val=0;
        for(int i=0;i<n-1;i++){
            val+=arr[i];
            if(val==arr2[i+1]){
            return true;
            }
        }
        val=0;
        for(int i=0;i<m-1;i++){
            val+=res[i];
            if(val==res2[i+1]){
            return true;
            }
        }
