class Solution {
    public int[][] reverseSubmatrix(int[][] grid, int x, int y, int k) {
        int n=grid.length;
        int m=grid[0].length;
        int result[][]=new int[n][m];
        int idx=x;
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                result[i][j]=grid[i][j];
            }
        }
        for(int i=x+k-1;i>=x;i--){
            for(int j=y;j<y+k;j++){
                result[idx][j]=grid[i][j];
            }
            idx++;
        }
        return result;
    }
}
