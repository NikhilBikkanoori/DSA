class Solution {
    int mod=(int)1e9+7;
    public int maxProductPath(int[][] grid) {
        int m=grid.length;
        int n=grid[0].length;
        long[][] dpmin=new long[m][n];
        long[][] dpmax=new long[m][n];
        dpmin[0][0]=grid[0][0];
        dpmax[0][0]=grid[0][0];
        for(int i=1;i<n;i++){
            dpmin[0][i]=(dpmin[0][i-1]*grid[0][i]);
            dpmax[0][i]=(dpmax[0][i-1]*grid[0][i]);
        }
        for(int i=1;i<m;i++){
            dpmin[i][0]=(dpmin[i-1][0]*grid[i][0]);
            dpmax[i][0]=(dpmax[i-1][0]*grid[i][0]);
        }
        for(int i=1;i<m;i++){
            for(int j=1;j<n;j++){
                int val=grid[i][j];
                long a=(dpmax[i-1][j]*val);
                long b=(dpmax[i][j-1]*val);
                long c=(dpmin[i-1][j]*val);
                long d=(dpmin[i][j-1]*val);
                dpmax[i][j]=Math.max(Math.max(a,b),Math.max(c,d));
                dpmin[i][j]=Math.min(Math.min(a,b),Math.min(c,d));
            }
        }
        int ans=(int)(dpmax[m-1][n-1]%mod);
        if(ans<0){
            return -1;
        }
        return ans;
    }
}
