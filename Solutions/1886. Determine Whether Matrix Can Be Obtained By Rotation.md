# 1886. Determine Whether Matrix Can Be Obtained By Rotation

class Solution {
    public boolean findRotation(int[][] mat, int[][] target) {
        int n=mat.length;
        int[][] nmat=new int[n][n];
        int cnt=0;
        while(cnt<=3){
            for(int i=n-1;i>=0;i--){
                for(int j=0;j<n;j++){
                    nmat[j][n-1-i]=mat[i][j];
                }
            }
            int flag=0;
            for(int i=0;i<n;i++){
                for(int j=0;j<n;j++){
                    if(nmat[i][j]!=target[i][j]){
                        flag++;
                    }
                    mat[i][j]=nmat[i][j];
                }
            }
            if(flag==0) return true;
            cnt++;
        }
        return false;
    }
}
