# 48. Rotate Image

**Difficulty:** Medium
**Topics:** Matrix,Array
**Date Solved:** 2026-05-02

class Solution {
    public void rotate(int[][] matrix) {
         int a=matrix.length;
        int g[][]=new int[a][a];
        int h=0;
        for(int i=a-1;i>=0;i--){
            for(int j=0;j<a;j++){
             g[j][h]=matrix[i][j];
            }
            h++;
        }
        for(int i=0;i<a;i++){
            for(int j=0;j<a;j++){
                matrix[i][j]=g[i][j];
            }
        
        }
    }
}
