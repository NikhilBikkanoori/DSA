# 796. Rotate String

class Solution {
    public boolean rotateString(String s, String goal) {
        int n=s.length();
        int a=0;
        while(n-->0){
            String res="";
            int b=a,m=s.length();
            while(m-->0){
                res+=s.charAt(b);
                b++;
                if(b==s.length()){
                    b=0;
                }
            }
            if(res.equals(goal)){
                return true;
            }
            a++;
        }
        return false;
    }
}
