# 788. Rotated Digits

**Difficulty:** Medium
**Topics:** Array,Math,Dynamic Programming
**Date Solved:** 2026-05-02

// Your code here
class Solution {
    public int rotatedDigits(int n) {
        int cnt=0;
        while(n>0){
            int x=n;
            int d=0,c=0;
            while(x>0){
                if(x%10==3||x%10==4||x%10==7){
                    d++;
                    break;
                }
                if(x%10==2||x%10==5||x%10==6||x%10==9){
                    c++;
                }
                x/=10;
            }
            if(d==0&&c>=1){
                cnt++;
            }
            n-=1;
        }
        return cnt;   
    }
}
