class Solution {
    public boolean areSimilar(int[][] mat, int k) {
        int n=mat.length;
        int m=mat[0].length;
        int cp[][]=new int[n][m];
        k=k%m;
        for(int i=0;i<n;i++){
            if(i%2==0){
                int idx=0;
                for(int j=k;j<m;j++){
                    cp[i][idx++]=mat[i][j];
                }
                for(int j=0;j<k;j++){
                    cp[i][idx++]=mat[i][j];
                }
            }else{
                int idx=k;
                for(int j=0;j<m;j++){
                    cp[i][idx++]=mat[i][j];
                    if(idx==m)
                    idx=0;
                }
            }
        }
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(cp[i][j]!=mat[i][j]){
                    return false;
                }
            }
        }
        return true;
    }
}
