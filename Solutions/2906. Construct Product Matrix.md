class Solution {
    public int[][] constructProductMatrix(int[][] grid) {
        int MOD = 12345;
        int n = grid.length;
        int m = grid[0].length;
        int[][] p = new int[n][m];
        long suf = 1;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = m - 1; j >= 0; j--) {
                p[i][j] = (int) suf;
                suf = (suf*grid[i][j]) % MOD;
            }
        }
        long pre=1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                p[i][j]=(int)((pre* p[i][j]) % MOD);
                pre=(pre*grid[i][j]) % MOD;
            }
        }
        return p;
    }
}
