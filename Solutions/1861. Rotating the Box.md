class Solution {
    public char[][] rotateTheBox(char[][] box) {
        int n=box.length;
        int m=box[0].length;
        char[][] res=new char[m][n];
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                res[j][i]=box[i][j];
            }
        }
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                System.out.print(res[j][i]+" ");
            }
            System.out.println();
        }
        for(int i=0;i<n;i++){
            int val=1;
            // while(val>0){
            //     val=0;
            //     for(int j=0;j<m-1;j++){
            //     System.out.print(res[j][i]+" ");
            //         if(res[j][i]=='#'&&res[j+1][i]=='.'){
            //             res[j][i]='.';
            //             res[j+1][i]='#';
            //             val++;
            //         }
            //     }
            // System.out.println();
            // }
            int r=0;
            while(val>0){
                val=0;
                for(int j=m-1;j>=1+r;j--){
                    if(res[j][i]=='.'&&res[j-1][i]=='#'){
                        res[j][i]='#';
                        res[j-1][i]='.';
                        val++;
                    }
                }
                r++;
            }
        }
        int k=0;
        char[][] res2=new char[m][n];
        for(int i=n-1;i>=0;i--){
            for(int j=0;j<m;j++){
                res2[j][k]=res[j][i];
            }
            k++;
        }
        return res2;
    }
}
