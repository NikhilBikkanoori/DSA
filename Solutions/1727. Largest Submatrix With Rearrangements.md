class Solution {
    public int largestSubmatrix(int[][] matrix) {
        int n=matrix.length;
        int m=matrix[0].length;
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(i==0)continue;
                if(matraix[i][j]!=0)matrix[i][j]+=matrix[i-1][j];
            }
        }
        int max=Integer.MIN_VALUE;
        for(int i=0;i<n;i++){
            Arrays.sort(matrix[i]);
            int h=0,b=0,area=0;
            if(matrix[i][m-1]>0){
                area=matrix[i][m-1];
                h=matrix[i][m-1];
                b=1;
            }
            for(int j=m-2;j>=0;j--){
                if(matrix[i][j]>0){
                    b++;
                    h=Math.min(h,matrix[i][j]);
                    area=Math.max(area,h*b);
                }
                else b=0;
            }

            max=Math.max(area,max);
        }

        return max;
    }
}
