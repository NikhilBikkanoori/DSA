// Your code here
class Solution {
    StringBuilder res=new StringBuilder();
    StringBuilder sb=new StringBuilder();
    int val=0;
    public String getHappyString(int n, int k) {
        res=new StringBuilder();
        sb=new StringBuilder();
        val=0;
        // char arr={'a','b','c'};
        ghs(n,k,0,0,0);
        return sb.toString();
    }
    public void ghs(int n,int k,int a,int b,int c){
        if (!sb.isEmpty()) return;
        
        if(n==0){
            val++;
            if(k==val&&sb.isEmpty()){
                sb=new StringBuilder(res);
                return;
            }
            System.out.println(res);
            return;
        }
        if(a!=1){
            res.append("a");
            ghs(n-1,k,1,0,0);
            res.deleteCharAt(res.length() - 1);
        }
        if(b!=1){
            res.append("b");
            ghs(n-1,k,0,1,0);
            res.deleteCharAt(res.length() - 1);
        }
        if(c!=1){
            res.append("c");
            ghs(n-1,k,0,0,1);
            res.deleteCharAt(res.length() - 1);
        }
    }
}
