class Solution {
    public boolean uniformArray(int[] nums1) {
        int odd=Integer.MAX_VALUE;
        int even=Integer.MAX_VALUE;
        int e=0,o=0;
        for(int i=0;i<nums1.length;i++){
            if(nums1[i]%2==0){
                if(even>nums1[i]){
                    even=nums1[i];
                }
                e++;
            }else{
                if(odd>nums1[i]){
                    odd=nums1[i];
                }
                o++;
            }
        }
        if(e==0||o==0){
            return true;
        }else{
            if(even-odd<1){
                return false;
            }else{
                return true;
            }
        }
    }
}
