class Solution {
    public int countCommas(int n) {
        int cnt=0;
        int m=n;
        while(n>0){
            n=n/10;
            cnt++;
        }
        int val=0;
        if(cnt>=4){
            val=1000;
        return m-val+1;
        }else{
            return val;
        }
        // if(cnt%3>=1){

        // }
        // return cnt/3-1;
    }
}
