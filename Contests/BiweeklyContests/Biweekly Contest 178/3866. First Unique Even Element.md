# 3866. First Unique Even Element

class Solution {
    public int firstUniqueEven(int[] nums) {
        int arr[]=new int[101];
        for(int i=0;i<nums.length;i++){
            arr[nums[i]]++;
        }
        for(int i=0;i<nums.length;i++){
            if(arr[nums[i]]==1&&nums[i]%2==0) return nums[i];
        }
        return -1;
    }
}
