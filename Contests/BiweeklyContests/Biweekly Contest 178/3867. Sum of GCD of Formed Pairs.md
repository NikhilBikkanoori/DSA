class Solution {
    public long gcdSum(int[] nums) {
        long gcd[]=new long[nums.length];
        // long gcd=0;
        int val=-1;
        for(int i=0;i<nums.length;i++){
            if(val<nums[i]){
                val=nums[i];
            }
            gcd[i]=gd(nums[i],val);
        }
        Arrays.sort(gcd);
        int n=nums.length/2;
        // if(nums.length%2==1){
        //     n+=1;
        // }
        int arr[]=new int[n];
        long res=0;
        int idx=0;
        for(int i=0;i<n;i++){
            res+=gd(gcd[i],gcd[nums.length-1-i]);
        }
        return res;
    }
    public long gd(long a,long b){
        while(b!=0){
            long temp=b;
            b=a%b;
            a=temp;
        }
        return (long)a;
    }
}
